<Section eyebrow="01 / Write" title="A test is a TypeScript file.">
  <p>
    Describe one user’s journey as an ordinary async function. Mantix runs
    it many times, on a schedule you choose, and records every request by
    its route.
  </p>
  <ul>
    <li>
      <code>rate.ramping</code> sets how many journeys start each second.
    </li>
    <li>
      <code>expect</code> checks the body with Zod, Valibot or Effect
      Schema.
    </li>
    <li>
      <code>thresholds</code> decide whether the run passes.
    </li>
  </ul>
  <Fragment slot="aside">

```ts title="examples/checkout.ts"
import { rate, scenario } from 'mantix'
import { z } from 'zod'

const Cart = z.object({
  id: z.string(),
  items: z.array(z.object({ sku: z.string(), price: z.number() })),
})

export default scenario({
  name: 'checkout',
  baseUrl: 'http://localhost:4200',
  load: rate.ramping({
    stages: [
      { duration: '10s', target: 50 }, // warm up
      { duration: '50s', target: 50 }, // hold
    ],
  }),
  thresholds: ['p99 < 250ms', 'errors < 1%'],
  iteration: async ({ http, vu }) => {
    const cartId = `cart-${vu}`
    await http.get('/products/:sku', { params: { sku: 'A1' } })
    await http.post('/cart/items', { json: { cartId, sku: 'A1', qty: 1 } })
    await http.get('/cart/:id', { params: { id: cartId }, expect: Cart })
    await http.post('/checkout', { json: { cartId } })
  },
})
```
